Beta diversity looks at the differences in group composition.

Load in Libraries and Set Working Directory

Load in Data

#filtered phyloseq -- could potentially see how results differ if I don't use the filtered version
physeq_filt <- readRDS("../results/exploratory/phyloseq_filtered.rds")
physeq <- readRDS("../results/phylogeny/phyloseq.rds") #going to stick with using unfiltered

Distance Metric

#bray-curtis
bray <- distance(
    physeq,
    method="bray"
)
#weighted unifrac
wunifrac <- distance(
    physeq,
    method="wunifrac"
)

Ordination

#could experiment with method parameter
ord_bc <- ordinate(
    physeq,
    method="PCoA", #principal coordinate analysis, similar to PCA
    distance=bray
)

ord_wuf <- ordinate(
    physeq,
    method="PCoA", #principal coordinate analysis, similar to PCA
    distance=wunifrac
)

Visualize

ord_bc_df <- data.frame(
  Sample = rownames(ord_bc$vectors),
  Axis.1 = ord_bc$vectors[, 1],
  Axis.2 = ord_bc$vectors[, 2]
)

ord_wuf_df <- data.frame(
  Sample = rownames(ord_wuf$vectors),
  Axis.1 = ord_wuf$vectors[, 1],
  Axis.2 = ord_wuf$vectors[, 2]
)

meta_df <- data.frame(sample_data(physeq))
meta_df$Sample <- rownames(meta_df)

ord_bc_df <- merge(ord_bc_df, meta_df, by = "Sample")
ord_wuf_df <- merge(ord_wuf_df, meta_df, by = "Sample")

ord_bc_plot <- ggplot(ord_bc_df, aes(x = Axis.1, y = Axis.2, color = Group)) +

    x = "PCoA 1",
    y = "PCoA 2",

Test Statistically

#bray curtis
#shows significance
adonis2(
    bray ~ Group,
    data=data.frame(sample_data(physeq))
)

    bray,
